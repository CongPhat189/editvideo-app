import { DarkTheme, DefaultTheme, Stack, ThemeProvider } from 'expo-router';
import { useCallback, useEffect, useReducer, useRef, useState } from 'react';

import { MyUserContext } from '@/config/Contexts';
import { palettes, ThemeContext } from '@/constants/theme';
import { MyUserReducer } from '@/reducers/reducers';
import { getThemePreference, saveThemePreference } from '@/services/appearance';

// Trang mở từ Sidebar không trượt thêm: nội dung tự trượt vào bằng hiệu ứng khép Sidebar
const drawerScreenOptions = ({ route }) => ({
  animation: route.params?.fromDrawer ? 'none' : 'fade',
  gestureEnabled: true,
});

export default function RootLayout() {
  const [user, dispatch] = useReducer(MyUserReducer, null);
  const [theme, setTheme] = useState('dark');
  const themeVersion = useRef(0);
  const token = user?.access_token;
  const colors = palettes[theme];

  useEffect(() => {
    if (!token) {
      setTheme('dark');
      return undefined;
    }
    let active = true;
    const requestedVersion = themeVersion.current;
    getThemePreference(token)
      .then((savedTheme) => {
        if (active && requestedVersion === themeVersion.current) setTheme(savedTheme);
      })
      .catch(() => {});
    return () => { active = false; };
  }, [token]);

  const updateTheme = useCallback((nextTheme) => {
    themeVersion.current += 1;
    setTheme(nextTheme);
    if (token) saveThemePreference(token, nextTheme).catch(() => {});
  }, [token]);

  return (
    <MyUserContext.Provider value={[user, dispatch]}>
      <ThemeContext.Provider value={{ theme, colors, setTheme: updateTheme }}>
        <ThemeProvider value={theme === 'dark' ? DarkTheme : DefaultTheme}>
          <Stack
            screenOptions={{
              headerShown: false,
              animation: 'slide_from_right',
              gestureEnabled: true,
              contentStyle: { backgroundColor: colors.backgroundDeep },
            }}
          >
            <Stack.Protected guard={!!user}>
              <Stack.Screen name="index" options={{ gestureEnabled: false }} />
              <Stack.Screen name="settings" options={{ presentation: 'fullScreenModal' }} />
              <Stack.Screen name="profile" options={{ presentation: 'fullScreenModal' }} />
              <Stack.Screen name="projects/new" options={drawerScreenOptions} />
              <Stack.Screen name="projects/[id]/index" options={drawerScreenOptions} />
              <Stack.Screen name="projects/[id]/schedule" />
              <Stack.Screen name="scheduled" options={drawerScreenOptions} />
              <Stack.Screen name="agents/index" options={drawerScreenOptions} />
              <Stack.Screen name="agents/new" />
            </Stack.Protected>
            <Stack.Protected guard={!user}>
              <Stack.Screen name="login" />
              <Stack.Screen name="register" />
            </Stack.Protected>
          </Stack>
        </ThemeProvider>
      </ThemeContext.Provider>
    </MyUserContext.Provider>
  );
}

const getScheduleAttachments = (schedule) => {
  const attachments = schedule.attachments ?? schedule.media ?? schedule.files;
  if (Array.isArray(attachments)) {
    return attachments.map((file, index) => ({
      ...file,
      id: file.id ?? `${schedule.id}-attachment-${index}`,
      url: file.url ?? file.uri ?? file.image_url,
      mime_type: file.mime_type ?? file.mimeType ?? 'image/jpeg',
      file_name: file.file_name ?? file.fileName ?? `Ảnh ${index + 1}`,
    }));
  }

  const imageUrls = schedule.image_urls ?? schedule.images ?? (schedule.image_url ? [schedule.image_url] : []);
  if (!Array.isArray(imageUrls)) return [];
  return imageUrls.map((image, index) => ({
    id: `${schedule.id}-image-${index}`,
    url: typeof image === 'string' ? image : image.url ?? image.uri,
    mime_type: typeof image === 'object' ? image.mime_type ?? image.mimeType ?? 'image/jpeg' : 'image/jpeg',
    file_name: typeof image === 'object' ? image.file_name ?? image.fileName ?? `Ảnh ${index + 1}` : `Ảnh ${index + 1}`,
  })).filter((image) => image.url);
};

function ScheduledContent({ menuButton }) {
  const colors = useThemeColors();
  const styles = useThemeStyles(createStyles);

  const [user] = use(MyUserContext);
  const token = user?.access_token;
  const [view, setView] = useState(() => {
    const today = new Date();
    return { year: today.getFullYear(), month: today.getMonth() };
  });
  const [selectedKey, setSelectedKey] = useState(() => toDayKey(new Date()));
  const [schedules, setSchedules] = useState(null);
  const [projectNames, setProjectNames] = useState({});
  const [selectedId, setSelectedId] = useState(null);
  const [error, setError] = useState('');

  useFocusEffect(
    useCallback(() => {
      if (!token) return undefined;
      let active = true;
      Promise.all([getSchedules(token), getProjects(token)])
        .then(([scheduleItems, projects]) => {
          if (!active) return;
          setSchedules(scheduleItems);
          setProjectNames(Object.fromEntries(projects.map((project) => [project.id, project.title])));
          setError('');
        })
        .catch((err) => {
          if (!active) return;
          setSchedules([]);
          setError(getErrorMessage(err));
        });
      return () => {
        active = false;
      };
    }, [token])
  );

  const markedKeys = useMemo(() => new Set((schedules ?? []).map(getScheduleDayKey)), [schedules]);
  const daySchedules = useMemo(
    () =>
      (schedules ?? [])
        .filter((schedule) => getScheduleDayKey(schedule) === selectedKey)
        .sort((a, b) => a.scheduled_at.localeCompare(b.scheduled_at)),
    [schedules, selectedKey]
  );
  const selected = daySchedules.find((schedule) => schedule.id === selectedId);
  const selectedAttachments = selected ? getScheduleAttachments(selected) : [];

  const changeMonth = (delta) => {
    setView(({ year, month }) => {
      const next = new Date(year, month + delta, 1);
      return { year: next.getFullYear(), month: next.getMonth() };
    });
  };

  const selectDay = (key) => {
    setSelectedKey(key);
    setSelectedId(null);
  };

  const projectName = selected ? projectNames[selected.conversation_id] : null;

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar style={colors.backgroundDeep === '#000000' ? 'light' : 'dark'} />
      <PageHeader title="Đã lên lịch" left={menuButton} />
      <ScrollView contentContainerStyle={styles.content} showsVerticalScrollIndicator={false}>
        <View style={styles.page}>
          <MonthCalendar
            year={view.year}
            month={view.month}
            selectedKey={selectedKey}
            markedKeys={markedKeys}
            onSelectDay={selectDay}
            onChangeMonth={changeMonth}
          />

          {error ? <Text style={styles.error}>{error}</Text> : null}
          {schedules === null ? <ActivityIndicator color={colors.text} style={styles.loading} /> : null}

          {schedules?.length === 0 && !error ? (
            <View style={styles.empty}>
              <Text style={styles.emptyTitle}>Chưa có lịch nào</Text>
              <Text style={styles.emptyBody}>Mở một dự án và đặt lịch ngay trong Chat, lịch sẽ hiện ở đây.</Text>
            </View>
          ) : null}

          {schedules?.length > 0 && daySchedules.length === 0 ? (
            <Text style={styles.noSchedule}>Không có lịch trong ngày này</Text>
          ) : null}

          {daySchedules.length > 0 ? (
            <View style={styles.card}>
              {daySchedules.map((schedule, index) => (
                <Pressable
                  key={schedule.id}
                  onPress={() => setSelectedId(schedule.id)}
                  style={({ pressed }) => [styles.row, pressed && styles.pressed]}
                  accessibilityRole="button"
                  accessibilityLabel={schedule.task_name}
                  aria-selected={schedule.id === selectedId}
                >
                  <View style={[styles.rowBody, index > 0 && styles.rowDivider]}>
                    <Text style={styles.rowTime}>{formatScheduleTime(schedule).slice(11)}</Text>
                    <Text style={styles.rowLabel} numberOfLines={1}>{schedule.task_name}</Text>
                    <ChevronRightIcon size={20} color={colors.textSecondary} />
                  </View>
                </Pressable>
              ))}
            </View>
          ) : null}

          {selected ? (
            <View style={styles.detail}>
              <DetailRow label="Tên" value={selected.task_name} />
              <DetailRow label="Thời gian" value={formatScheduleTime(selected)} />
              {selected.platforms.length > 0 ? (
                <DetailRow label="Nền tảng" value={selected.platforms.join(', ')} />
              ) : null}
              {selected.content || selectedAttachments.length > 0 ? (
                <View style={styles.contentSection}>
                  <Text style={styles.detailLabel}>Nội dung</Text>
                  {selected.content ? <Text style={styles.detailValue} selectable>{selected.content}</Text> : null}
                  {selectedAttachments.length > 0 ? <MessageAttachments attachments={selectedAttachments} /> : null}
                </View>
              ) : null}
              <Pressable
                onPress={() => router.replace(`/projects/${selected.conversation_id}`)}
                disabled={!projectName}
                style={({ pressed }) => [styles.projectRow, pressed && styles.pressed]}
                accessibilityRole="button"
                accessibilityLabel={`Mở dự án ${projectName ?? ''}`}
              >
                <View style={styles.projectText}>
                  <Text style={styles.detailLabel}>Dự án</Text>
                  <Text style={styles.detailValue} numberOfLines={1}>
                    {projectName || 'Không tìm thấy dự án'}
                  </Text>
                </View>
                {projectName ? <ChevronRightIcon size={20} color={colors.textSecondary} /> : null}
              </Pressable>
            </View>
          ) : null}
        </View>
      </ScrollView>
    </SafeAreaView>
  );
}

import { useFocusEffect } from 'expo-router';
import { StatusBar } from 'expo-status-bar';
import { use, useCallback, useState } from 'react';
import {
  ActivityIndicator,
  Alert,
  KeyboardAvoidingView,
  LayoutAnimation,
  Platform,
  Pressable,
  RefreshControl,
  ScrollView,
  StyleSheet,
  Text,
  View,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';

import AgentForm from '@/components/AgentForm';
import AuthButton from '@/components/AuthButton';
import DrawerScreen from '@/components/DrawerScreen';
import {
  AgentIcon,
  PencilIcon,
  PluginIcon,
  TrashIcon,
} from '@/components/HomeIcons';
import PageHeader from '@/components/PageHeader';
import { MyUserContext } from '@/config/Contexts';
import { useThemeColors, useThemeStyles } from '@/constants/theme';
import {
  deleteAgent,
  getAgentErrorMessage,
  getAgentsByUser,
  testSavedAgentConnection,
} from '@/services/agents';

export default function Agents() {
  return (
    <DrawerScreen>
      {(menuButton) => <AgentsContent menuButton={menuButton} />}
    </DrawerScreen>
  );
}

function AgentsContent({ menuButton }) {
  const colors = useThemeColors();
  const styles = useThemeStyles(createStyles);

  const [user] = use(MyUserContext);
  const token = user?.access_token;
  const userId = user?.id;

  const [agents, setAgents] = useState(null);
  const [error, setError] = useState('');
  const [refreshing, setRefreshing] = useState(false);
  const [formOpen, setFormOpen] = useState(false);
  const [editingAgent, setEditingAgent] = useState(null);

  // Lưu trạng thái test kết nối cho từng agent: { [agentId]: { ok: bool, message: str } }
  const [testResults, setTestResults] = useState({});
  const [testingId, setTestingId] = useState(null);

  const loadAgents = useCallback(
    async (isPullRefresh = false) => {
      if (!token) return;
      if (isPullRefresh) setRefreshing(true);

      try {
        const data = await getAgentsByUser(token, userId);
        setAgents(data || []);
        setError('');
      } catch (err) {
        setError(getAgentErrorMessage(err));
      } finally {
        setRefreshing(false);
      }
    },
    [token, userId]
  );

  useFocusEffect(
    useCallback(() => {
      loadAgents();
    }, [loadAgents])
  );

  const toggleForm = () => {
    LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
    setEditingAgent(null);
    setFormOpen((open) => !open);
  };

  const handleEditAgent = (agent) => {
    LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
    setEditingAgent(agent);
    setFormOpen(true);
  };

  const handleDeleteAgent = (agent) => {
    Alert.alert(
      'Xác nhận xóa',
      `Bạn có chắc chắn muốn xóa agent "${agent.name}" không?`,
      [
        { text: 'Hủy', style: 'cancel' },
        {
          text: 'Xóa',
          style: 'destructive',
          onPress: async () => {
            try {
              await deleteAgent(token, agent.id);
              await loadAgents();
            } catch (err) {
              Alert.alert('Lỗi xóa Agent', getAgentErrorMessage(err));
            }
          },
        },
      ]
    );
  };

  const handleTestAgent = async (agent) => {
    setTestingId(agent.id);
    setTestResults((prev) => ({ ...prev, [agent.id]: null }));

    try {
      const res = await testSavedAgentConnection(token, agent.id);
      LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
      setTestResults((prev) => ({
        ...prev,
        [agent.id]: {
          ok: res.ok,
          message: res.ok ? 'Kết nối GoClaw thành công' : (res.message || 'Không kết nối được GoClaw'),
        },
      }));
    } catch (err) {
      LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
      setTestResults((prev) => ({
        ...prev,
        [agent.id]: {
          ok: false,
          message: getAgentErrorMessage(err),
        },
      }));
    } finally {
      setTestingId(null);
    }
  };

  const handleSaved = async () => {
    await loadAgents();
    LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
    setFormOpen(false);
    setEditingAgent(null);
  };

  const handleCancelForm = () => {
    LayoutAnimation.configureNext(LayoutAnimation.Presets.easeInEaseOut);
    setFormOpen(false);
    setEditingAgent(null);
  };

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar style={colors.backgroundDeep === '#000000' ? 'light' : 'dark'} />
      <PageHeader title="Agent" left={menuButton} />

      <KeyboardAvoidingView
        style={styles.flex}
        behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      >
        <ScrollView
          contentContainerStyle={styles.content}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}
          refreshControl={
            <RefreshControl
              refreshing={refreshing}
              onRefresh={() => loadAgents(true)}
              tintColor={colors.text}
              colors={[colors.text]}
            />
          }
        >
          <View style={styles.page}>
            {error ? <Text style={styles.error}>{error}</Text> : null}

            {agents === null && !error ? (
              <ActivityIndicator color={colors.text} style={styles.loading} />
            ) : null}

            {/* Form Thêm mới / Chỉnh sửa */}
            {formOpen ? (
              <View style={styles.formContainer}>
                <AgentForm
                  agent={editingAgent}
                  onSaved={handleSaved}
                  onCancel={handleCancelForm}
                />
              </View>
            ) : (
              <View style={styles.topAction}>
                <AuthButton title="+ Thêm Agent" onPress={toggleForm} />
              </View>
            )}

            {/* Trạng thái trống */}
            {agents?.length === 0 && !formOpen ? (
              <View style={styles.emptyContainer}>
                <View style={styles.emptyIconBox}>
                  <AgentIcon size={36} color={colors.textMuted} />
                </View>
                <Text style={styles.emptyTitle}>Chưa có Agent nào</Text>
                <Text style={styles.emptySubtitle}>
                  Thêm máy chủ GoClaw Agent để bắt đầu tự động hóa quy trình dựng và xuất bản video.
                </Text>
              </View>
            ) : null}

            {/* Danh sách Agent Cards */}
            {agents?.length > 0 ? (
              <View style={styles.list}>
                {agents.map((agent) => {
                  const isTesting = testingId === agent.id;
                  const testStatus = testResults[agent.id];

                  return (
                    <View key={agent.id} style={styles.agentCard}>
                      {/* Header của Card */}
                      <View style={styles.cardHeader}>
                        <View style={styles.cardAvatar}>
                          <AgentIcon size={22} color={colors.primary} />
                        </View>
                        <View style={styles.cardTitleBox}>
                          <Text style={styles.cardTitle} numberOfLines={1}>
                            {agent.name}
                          </Text>
                          <View style={styles.domainRow}>
                            <PluginIcon size={13} color={colors.textMuted} />
                            <Text style={styles.domainText} numberOfLines={1}>
                              {agent.domain || 'Chưa có domain'}
                            </Text>
                          </View>
                        </View>

                        {/* Nút Sửa & Xóa */}
                        <View style={styles.cardControls}>
                          <Pressable
                            style={({ pressed }) => [styles.ctrlBtn, pressed && styles.pressed]}
                            onPress={() => handleEditAgent(agent)}
                            accessibilityLabel="Sửa"
                            hitSlop={8}
                          >
                            <PencilIcon size={17} color={colors.textSecondary} />
                          </Pressable>
                          <Pressable
                            style={({ pressed }) => [styles.ctrlBtn, pressed && styles.pressed]}
                            onPress={() => handleDeleteAgent(agent)}
                            accessibilityLabel="Xóa"
                            hitSlop={8}
                          >
                            <TrashIcon size={17} color={colors.error} />
                          </Pressable>
                        </View>
                      </View>

                      {/* Thông tin Key bảo mật */}
                      <View style={styles.keyInfoRow}>
                        <View style={styles.keyBadge}>
                          <Text style={styles.keyBadgeLabel}>API Key:</Text>
                          <Text style={styles.keyBadgeVal}>
                            {agent.has_api_key ? agent.masked_api_key : 'Chưa cấu hình'}
                          </Text>
                        </View>
                        {agent.has_agent_key ? (
                          <View style={styles.keyBadge}>
                            <Text style={styles.keyBadgeLabel}>Agent Key:</Text>
                            <Text style={styles.keyBadgeVal}>{agent.masked_agent_key}</Text>
                          </View>
                        ) : null}
                      </View>

                      {/* Trạng thái Test Kết Nối (Pill thanh lịch, không dùng icon tick to giả tạo) */}
                      {testStatus ? (
                        <View
                          style={[
                            styles.statusPill,
                            testStatus.ok ? styles.statusPillOk : styles.statusPillFail,
                          ]}
                        >
                          <View
                            style={[
                              styles.statusDot,
                              testStatus.ok ? styles.statusDotOk : styles.statusDotFail,
                            ]}
                          />
                          <Text
                            style={[
                              styles.statusText,
                              testStatus.ok ? styles.statusTextOk : styles.statusTextFail,
                            ]}
                          >
                            {testStatus.message}
                          </Text>
                        </View>
                      ) : null}

                      {/* Nút Test kết nối của thẻ */}
                      <Pressable
                        style={({ pressed }) => [
                          styles.testCardBtn,
                          pressed && styles.pressed,
                          isTesting && styles.btnDisabled,
                        ]}
                        onPress={() => handleTestAgent(agent)}
                        disabled={isTesting}
                      >
                        {isTesting ? (
                          <ActivityIndicator size="small" color={colors.text} />
                        ) : (
                          <Text style={styles.testCardBtnText}>Kiểm tra kết nối</Text>
                        )}
                      </Pressable>
                    </View>
                  );
                })}
              </View>
            ) : null}
          </View>
        </ScrollView>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

const MessageBubble = ({ message, showScheduleButton, onLongPress, onSchedule }) => {
  const colors = useThemeColors();
  const styles = useThemeStyles(createStyles);

  const attachments = message.attachments ?? [];
  if (message.role === 'user') {
    return (
      <View style={styles.userMessage}>
        {attachments.length > 0 ? <MessageAttachments attachments={attachments} align="flex-end" /> : null}
        {message.content ? (
          <View style={styles.userBubble}>
            <Text style={styles.messageText} selectable>{message.content}</Text>
          </View>
        ) : null}
      </View>
    );
  }
  if (IN_PROGRESS.includes(message.status)) {
    return (
      <View style={styles.replyPending}>
        <ActivityIndicator size="small" color={colors.textSecondary} />
        <Text style={styles.pendingText}>Agent đang trả lời…</Text>

    return <Text style={styles.failedText}>{message.error_message || 'Agent không trả lời được'}</Text>;

      <Pressable
        onLongPress={onLongPress}

function ChatContent({ id, menuButton }) {
  const colors = useThemeColors();
  const styles = useThemeStyles(createStyles);

  const [user] = use(MyUserContext);
  const token = user?.access_token;
  const [project, setProject] = useState(null);
  const [messages, setMessages] = useState([]);
  const [nextBefore, setNextBefore] = useState(null);
  const [loading, setLoading] = useState(true);
  const [loadError, setLoadError] = useState('');
  const [text, setText] = useState('');
  const [files, setFiles] = useState([]);
  const [sending, setSending] = useState(false);
  const [sendError, setSendError] = useState('');
  const [scheduleActionMessageId, setScheduleActionMessageId] = useState(null);
  const loadingOlder = useRef(false);
  const inputRef = useRef(null);

  const loadMessages = useCallback(async () => {
    const page = await getMessages(token, id);
    setMessages(page.items);
    setNextBefore(page.next_before);
  }, [token, id]);

  // Chỉ đọc dự án đã có và ghi lại lần mở, không tạo dự án mới
  useEffect(() => {
    if (!token) return;
    getProject(token, id)
      .then((item) => {
        setProject(item);
        return markProjectOpened(token, item.id);
      })
      .catch((error) => setLoadError(getErrorMessage(error)));
  }, [token, id]);

  useFocusEffect(
    useCallback(() => {
      if (!token) return;
      loadMessages()
        .catch((error) => setLoadError(getErrorMessage(error)))
        .finally(() => setLoading(false));
    }, [token, loadMessages])
  );

  // Câu trả lời đang chạy trong nền (vd lịch vừa đến giờ): hỏi lại đến khi xong
  const hasReplyInProgress = !sending && messages.some((message) => IN_PROGRESS.includes(message.status));
  useEffect(() => {
    if (!hasReplyInProgress) return undefined;
    const timer = setInterval(() => {
      loadMessages().catch(() => {});
    }, POLL_INTERVAL);
    return () => clearInterval(timer);
  }, [hasReplyInProgress, loadMessages]);

  const loadOlder = async () => {
    if (!nextBefore || loadingOlder.current) return;
    loadingOlder.current = true;
    try {
      const page = await getMessages(token, id, nextBefore);
      setMessages((current) => [...current, ...page.items]);
      setNextBefore(page.next_before);
    } catch (error) {
      setSendError(getErrorMessage(error));
    } finally {
      loadingOlder.current = false;
    }
  };

  const handlePickFiles = async () => {
    const remaining = MAX_FILES - files.length;
    if (remaining <= 0) {
      setSendError(`Mỗi tin gửi tối đa ${MAX_FILES} file`);
      return;
    }
    setSendError('');
    try {
      const picked = await pickMedia(remaining);
      setFiles((current) => {
        const newFiles = picked.files.filter((file) => !current.some((item) => item.uri === file.uri));
        return [...current, ...newFiles];
      });
      if (picked.error) setSendError(picked.error);
    } catch {
      setSendError('Không mở được thư viện ảnh');
    }
  };

  const handleSend = async () => {
    const content = text.trim();
    if ((!content && files.length === 0) || sending) return;

    const sentFiles = files;
    const localUserMessage = {
      id: 'local-user',
      role: 'user',
      content,
      status: 'completed',
      attachments: sentFiles.map((file) => ({
        id: file.uri,
        file_name: file.fileName,
        mime_type: file.mimeType,
        url: file.uri,
      })),
    };
    const localReply = { id: 'local-reply', role: 'assistant', status: 'processing' };
    setMessages((current) => [localReply, localUserMessage, ...current]);
    setText('');
    setFiles([]);
    setSendError('');
    setSending(true);
    try {
      const result = sentFiles.length > 0
        ? await sendMessageWithFiles(token, id, content, sentFiles)
        : await sendMessage(token, id, content);
      setMessages((current) => [
        ...(result.assistant_message ? [result.assistant_message] : []),
        result.user_message,
        ...current.filter((message) => !message.id.startsWith('local-')),
      ]);
    } catch (error) {
      setMessages((current) => current.filter((message) => !message.id.startsWith('local-')));
      setText(content);
      setFiles(sentFiles);
      setSendError(getErrorMessage(error));
    } finally {
      setSending(false);
    }
  };

  const canSend = (text.trim().length > 0 || files.length > 0) && !sending;

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar style={colors.backgroundDeep === '#000000' ? 'light' : 'dark'} />
      <PageHeader title={project?.title || ''} left={menuButton} />
      <KeyboardAvoidingView
        style={styles.flex}
        behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      >
        {loading ? (
          <View style={styles.center}>
            <ActivityIndicator color={colors.text} />
          </View>
        ) : null}
        {!loading && loadError ? (
          <View style={styles.center}>
            <Text style={styles.failedText}>{loadError}</Text>
          </View>
        ) : null}
        {!loading && !loadError && messages.length === 0 ? (
          <View style={styles.center}>
            <View style={styles.emptyState}>
              <View style={styles.emptyIcon}><AgentIcon size={24} /></View>
              <Text style={styles.emptyTitle}>Bắt đầu cuộc trò chuyện</Text>
              <Text style={styles.emptyText}>Mô tả video bạn muốn tạo. Bạn cũng có thể gửi ảnh hoặc video để Agent tham khảo.</Text>
            </View>
          </View>
        ) : null}
        {!loading && !loadError && messages.length > 0 ? (
          <FlatList
            style={styles.flex}
            inverted
            data={messages}
            keyExtractor={(message) => message.id}
            renderItem={({ item }) => (
              <MessageBubble
                message={item}
                showScheduleButton={String(scheduleActionMessageId) === String(item.id)}
                onLongPress={() => setScheduleActionMessageId(item.id)}
                onSchedule={() => router.push({
                  pathname: '/projects/[id]/schedule',
                  params: { id, contentMessageId: item.id },
                })}
              />
            )}
            contentContainerStyle={styles.messages}
            onEndReached={loadOlder}
            onEndReachedThreshold={0.3}
            keyboardShouldPersistTaps="handled"
          />
        ) : null}

        {sendError ? <Text style={styles.sendError}>{sendError}</Text> : null}
        {files.length > 0 ? (
          <View>
            <PendingAttachments
              files={files}
              onRemove={(uri) => setFiles((current) => current.filter((file) => file.uri !== uri))}
            />
          </View>
        ) : null}
        <View style={styles.composerWrap}>
          <View style={styles.composer}>
            <TextInput
              ref={inputRef}
              style={styles.input}
              value={text}
              onChangeText={setText}
              placeholder="Nhập tin nhắn"
              placeholderTextColor={colors.textMuted}
              selectionColor={colors.primary}
              accessibilityLabel="Nhập tin nhắn"
              maxLength={MAX_MESSAGE_LENGTH}
              multiline
            />
            <View style={styles.composerToolbar}>
              <View style={styles.composerActions}>
                <Pressable
                  onPress={handlePickFiles}
                  style={({ pressed }) => [styles.actionButton, pressed && styles.pressed]}
                  accessibilityRole="button"
                  accessibilityLabel="Thêm ảnh hoặc video"
                >
                  <PlusIcon size={20} />
                </Pressable>
                <Pressable
                  onPress={() => inputRef.current?.focus()}
                  disabled={sending}
                  style={({ pressed }) => [styles.actionButton, pressed && styles.pressed]}
                  accessibilityRole="button"
                  accessibilityLabel="Mở bàn phím để nhập bằng giọng nói"
                  accessibilityState={{ disabled: sending }}
                >
                  <MicIcon size={20} />
                </Pressable>
              </View>
              <Pressable
                onPress={handleSend}
                disabled={!canSend}
                style={({ pressed }) => [styles.sendButton, !canSend && styles.sendButtonDisabled, pressed && styles.pressed]}
                accessibilityRole="button"
                accessibilityLabel="Gửi"
                accessibilityState={{ disabled: !canSend }}
              >
                <SendIcon size={21} color={canSend ? colors.onPrimary : colors.textMuted} />
              </Pressable>
            </View>
          </View>
          <Text style={styles.composerHint}>AI có thể mắc lỗi. Hãy kiểm tra lại thông tin quan trọng.</Text>
        </View>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );

const getConversationMessages = async (token, conversationId) => {
  const messages = [];
  let before;
  for (let pageNumber = 0; pageNumber < MAX_MESSAGE_PAGES; pageNumber += 1) {
    const page = await getMessages(token, conversationId, before);
    messages.push(...page.items);
    if (!page.next_before) break;
    before = page.next_before;
  }
  return messages;
};

const getMessageLabel = (message) => {
  const text = message.content?.trim();
  if (text) return text;
  const attachments = message.attachments ?? [];
  return attachments.length ? `Nội dung có ${attachments.length} tệp đính kèm` : 'Nội dung không có văn bản';
};

import { router, useLocalSearchParams, usePathname } from 'expo-router';
import { useCallback, useEffect, useState } from 'react';
import { Keyboard } from 'react-native';

import MenuButton from '@/components/MenuButton';
import PushDrawer from '@/components/PushDrawer';
import Sidebar from '@/components/Sidebar';

const CLOSE_AFTER_PAGE_OPENS = 300;

// Khung chung của các trang mở từ Sidebar: bấm một mục thì Sidebar khép lại và nội dung trang đó trượt vào.
// root: Trang chủ giữ lại trong stack, các trang khác thay thế nhau để stack không dài thêm.
export default function DrawerScreen({ root = false, children }) {
  const { fromDrawer } = useLocalSearchParams();
  const pathname = usePathname();
  const [open, setOpen] = useState(fromDrawer === '1');
  const close = useCallback(() => setOpen(false), []);

  // Mở từ Sidebar: bắt đầu ở tư thế Sidebar đang mở rồi khép lại
  useEffect(() => {
    if (!open) return undefined;
    const frame = requestAnimationFrame(close);
    return () => cancelAnimationFrame(frame);
  }, []);

  const openMenu = () => {
    Keyboard.dismiss();
    setOpen(true);
  };

  const navigate = (href) => {
    if (href === pathname) {
      close();
      return;
    }
    const target = { pathname: href, params: { fromDrawer: '1' } };
    if (root) {
      router.push(target);
      // Khép khi trang mới đã che Trang chủ, để lúc quay về Sidebar ở đây không còn mở
      setTimeout(close, CLOSE_AFTER_PAGE_OPENS);
    } else {
      router.replace(target);
    }
  };

  return (
    <PushDrawer
      open={open}
      onClose={close}
      drawer={<Sidebar onClose={close} onNavigate={navigate} />}
    >
      {children(<MenuButton expanded={open} onPress={openMenu} />)}
    </PushDrawer>
  );
}

export const toDayKey = (date) => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;

const buildCells = (year, month) => {
  const firstWeekday = (new Date(year, month, 1).getDay() + 6) % 7; // Tuần bắt đầu từ thứ Hai
  const dayCount = new Date(year, month + 1, 0).getDate();
  const cells = Array.from({ length: firstWeekday }, () => null);
  for (let day = 1; day <= dayCount; day += 1) cells.push(day);
  while (cells.length % 7 !== 0) cells.push(null);
  return cells;
};

export default function MonthCalendar({ year, month, selectedKey, minKey, markedKeys, onSelectDay, onChangeMonth }) {

export default function PushDrawer({ open, onClose, drawer, children }) {
  const styles = useThemeStyles(createStyles);

  const { width } = useWindowDimensions();
  const drawerWidth = Math.min(width * WIDTH_RATIO, MAX_WIDTH);
  const reducedMotion = useReducedMotion();
  const progress = useRef(new Animated.Value(open ? 1 : 0)).current;
  const [mounted, setMounted] = useState(open);

  if (open && !mounted) setMounted(true);

  useEffect(() => {
    const duration = open ? OPEN_DURATION : CLOSE_DURATION;
    Animated.timing(progress, {
      toValue: open ? 1 : 0,
      duration: reducedMotion ? 0 : duration,
      easing: Easing.out(Easing.cubic),
      useNativeDriver: true,
    }).start(({ finished }) => {
      if (finished && !open) setMounted(false);

      if (!open) return undefined;
      const subscription = BackHandler.addEventListener('hardwareBackPress', () => {
        onClose();
        return true;
      });
      return () => subscription.remove();
    }, [open, onClose])
  );

  const translateX = progress.interpolate({ inputRange: [0, 1], outputRange: [0, drawerWidth] });

// Giữ vị trí bánh xe ngoài state vì nó đổi theo từng khung hình khi vuốt. onChange phải là hàm ổn định
const createWheel = ({ count, initialValue, onChange, setValue }) => {
  const startTop = START_ROW + initialValue - CENTER_ROW;
  const scrollY = new Animated.Value(startTop * ROW_HEIGHT);
  let top = startTop;
  let value = initialValue;

  return {
    scrollY,
    startTop,
    nextOffset: (delta) => (top + delta) * ROW_HEIGHT,
    onScroll: Animated.event([{ nativeEvent: { contentOffset: { y: scrollY } } }], {
      useNativeDriver: true,
      listener: (event) => {
        // Android có thể báo vị trí 0 lúc mới dựng, người dùng thì không vuốt tới được đầu danh sách
        const offsetY = event.nativeEvent.contentOffset.y;
        if (offsetY === 0) return;
        top = Math.round(offsetY / ROW_HEIGHT);
        const next = (top + CENTER_ROW) % count;
        if (next === value) return;
        value = next;
        setValue(next);
        onChange(next);
      },
    }),
  };
};

export default function TimeWheel({ count, initialValue, label, onChange }) {
  const styles = useThemeStyles(createStyles);

  const listRef = useRef(null);
  const [value, setValue] = useState(initialValue);
  const [wheel] = useState(() => createWheel({ count, initialValue, onChange, setValue }));
  const { scrollY } = wheel;

  const renderItem = useCallback(
    ({ item }) => <WheelRow row={item} count={count} scrollY={scrollY} />,
    [count, scrollY]
  );

  const handleAccessibilityAction = ({ nativeEvent }) => {
    const delta = nativeEvent.actionName === 'increment' ? 1 : -1;
    listRef.current?.scrollToOffset({ offset: wheel.nextOffset(delta), animated: true });
  };

  return (
    <View
      style={styles.wheel}
      accessible
      accessibilityRole="adjustable"
      accessibilityLabel={label}
      accessibilityValue={{ text: pad(value) }}
      accessibilityActions={[{ name: 'increment' }, { name: 'decrement' }]}
      onAccessibilityAction={handleAccessibilityAction}
    >
      <Animated.FlatList
        ref={listRef}
        data={ROWS}
        renderItem={renderItem}
        keyExtractor={keyExtractor}
        getItemLayout={getItemLayout}
        initialScrollIndex={wheel.startTop}
        initialNumToRender={VISIBLE_ROWS + 2}
        windowSize={11}

        overScrollMode="never"
        showsVerticalScrollIndicator={false}
        scrollEventThrottle={16}
        onScroll={wheel.onScroll}
      />
    </View>
  );
}

export const isImage = (mimeType) => mimeType?.startsWith('image/');
export const isVideo = (mimeType) => mimeType?.startsWith('video/');

const toFile = (asset, index) => {
  const mimeType = asset.mimeType ?? (asset.type === 'video' ? 'video/mp4' : 'image/jpeg');
  const extension = mimeType.split('/')[1];
  return {
    uri: asset.uri,
    mimeType,
    fileName: asset.fileName ?? `file-${Date.now()}-${index}.${extension}`,
    fileSize: asset.fileSize,
  };
};

export const pickMedia = async (remaining) => {
  const permission = await ImagePicker.requestMediaLibraryPermissionsAsync();
  if (!permission.granted) {
    return { files: [], error: 'Hãy cho phép app truy cập thư viện ảnh trong Cài đặt của điện thoại' };
  }

  const result = await ImagePicker.launchImageLibraryAsync({
    mediaTypes: ['images', 'videos'],
    allowsMultipleSelection: true,
    selectionLimit: remaining,
  });
  if (result.canceled) return { files: [], error: '' };

  const picked = result.assets.map(toFile);
  const accepted = picked.filter((file) => !file.fileSize || file.fileSize <= MAX_FILE_BYTES);
  const files = accepted.slice(0, remaining);
  if (accepted.length < picked.length) {
    return { files, error: 'Có file lớn hơn 50 MB nên chưa được thêm' };
  }
  if (files.length < accepted.length) {
    return { files, error: `Mỗi tin gửi tối đa ${MAX_FILES} file` };
  }
  return { files, error: '' };
};

export const validateScheduleTime = (text) => {
  const match = INPUT_FORMAT.exec(text);
  if (!match) return 'Nhập theo dạng 05/11/2026 20:00';
  const [day, month, year, hour, minute] = match.slice(1).map(Number);
  const date = new Date(Date.UTC(year, month - 1, day, hour, minute));
  const isRealDate =
    date.getUTCDate() === day &&
    date.getUTCMonth() === month - 1 &&
    date.getUTCHours() === hour &&
    date.getUTCMinutes() === minute;
  if (!isRealDate) return 'Ngày giờ không hợp lệ';
  if (date.getTime() - TIMEZONE_OFFSET_HOURS * 3600000 <= Date.now()) return 'Thời gian phải ở tương lai';
  return null;
};
