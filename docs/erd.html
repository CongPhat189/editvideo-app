    users {
        uuid id PK
        varchar full_name
        varchar phone UK "E.164"
        text password_hash "Argon2id"
        int token_version "đăng xuất thì tăng"
        text avatar_url
        varchar status "active|suspended|deleted"
        timestamptz created_at
        timestamptz updated_at
        timestamptz last_login_at
        timestamptz deleted_at
    }

    user_settings {
        uuid id PK
        uuid user_id FK,UK
        varchar theme "light|dark|system"
        varchar language
        varchar timezone "tên IANA"
        boolean notification_enabled
        jsonb settings_json
        timestamptz created_at
        timestamptz updated_at
    }

    agents {
        uuid id PK
        varchar name
        text key "API key - mã hoá khi lưu"
        varchar domain "vd: https://goclaw.abc.com"
        timestamptz created_at
        timestamptz updated_at
    }

    conversations {
        uuid id PK
        uuid user_id FK
        uuid agent_id FK
        varchar title
        varchar goclaw_session_id UK "nullable"
        varchar status "active|archived|deleted"
        int message_count
        timestamptz last_message_at
        timestamptz pinned_at
        timestamptz created_at
        timestamptz updated_at
        timestamptz archived_at
        timestamptz deleted_at
    }

    messages {
        uuid id PK
        uuid conversation_id FK
        uuid user_id FK
        varchar role "user|assistant|system|tool"
        text content
        varchar content_type "text|image|file|audio|video"
        varchar goclaw_message_id
        uuid reply_to_message_id FK
        varchar status "pending..cancelled"
        uuid idempotency_key UK "per user"
        text error_message
        jsonb metadata "tokens, model, tools"
        timestamptz favorited_at
        timestamptz created_at
        timestamptz updated_at
    }

    message_attachments {
        uuid id PK
        uuid message_id FK
        varchar file_name
        text file_path UK
        text file_url
        varchar mime_type
        bigint file_size
        timestamptz created_at
    }

    scheduled_tasks {
        uuid id PK
        uuid user_id FK
        uuid conversation_id FK "nullable"
        uuid agent_id FK "nullable"
        varchar task_name
        varchar task_type "send_message|execute_agent|..."
        jsonb payload
        timestamptz scheduled_at
        varchar timezone
        text recurrence_rule "RRULE"
        varchar status "scheduled..retrying"
        int attempts
        int max_attempts
        timestamptz next_retry_at
        text last_error
        varchar locked_by
        timestamptz locked_at
        timestamptz executed_at
        timestamptz completed_at
        timestamptz created_at
        timestamptz updated_at
    }

    scheduled_task_logs {
        uuid id PK
        uuid task_id FK
        int attempt_number
        varchar status
        jsonb request_payload
        jsonb response_payload
        text error_message
        timestamptz started_at
        timestamptz finished_at
        timestamptz created_at
    }
